```javascript
const express = require('express');
const dotenv = require('dotenv');
const cors = require('cors');
const helmet = require('helmet');
const morgan = require('morgan');
const rateLimit = require('express-rate-limit');
const connectDB = require('./config/db');
const errorHandler = require('./middleware/errorHandler');

dotenv.config();
connectDB();

const app = express();

app.use(helmet());
app.use(cors());

const limiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 100,
  message: { success: false, message: 'Too many requests, please try again later.' }
});
app.use('/api/', limiter);

app.use(express.json());
app.use(express.urlencoded({ extended: true }));

if (process.env.NODE_ENV === 'development') {
  app.use(morgan('dev'));
}

// Routes
app.use('/api/auth', require('./routes/auth'));
app.use('/api/matches', require('./routes/matches'));
app.use('/api/players', require('./routes/players'));
app.use('/api/teams', require('./routes/teams'));
app.use('/api/saved-matches', require('./routes/savedMatches'));

app.get('/api/health', (req, res) => {
  res.json({ success: true, message: 'Server is running', timestamp: new Date().toISOString() });
});

app.use(errorHandler);

const PORT = process.env.PORT || 5000;
app.listen(PORT, () => {
  console.log(`🏏 Apna Cricket Server running on port ${PORT}`);
});

module.exports = app;
```

```javascript
const bcrypt = require('bcryptjs');

const userSchema = new mongoose.Schema({
  name: { type: String, required: [true, 'Name is required'], trim: true, maxlength: 50 },
  email: { type: String, required: [true, 'Email is required'], unique: true, lowercase: true, trim: true },
  password: { type: String, required: [true, 'Password is required'], minlength: 6, select: false },

userSchema.pre('save', async function(next) {
  if (!this.isModified('password')) return next();
  const salt = await bcrypt.genSalt(10);
  this.password = await bcrypt.hash(this.password, salt);
  next();
});

userSchema.methods.comparePassword = async function(enteredPassword) {
  return await bcrypt.compare(enteredPassword, this.password);
};

module.exports = mongoose.model('User', userSchema);
```

```javascript
  matchType: { type: String, enum: ['T20', 'ODI', 'TEST', 'T10', 'T100'], default: 'T20' },
  league: { type: String, required: true },
  venue: { type: String, default: '' },
  startTime: { type: Date, required: true },
  status: { type: String, enum: ['upcoming', 'live', 'completed', 'cancelled'], default: 'upcoming' },
  pitchReport: { type: String, default: 'Balanced pitch.' },
  weather: { type: String, default: 'Clear sky' },
  formats: [{ type: String, enum: ['Mega', 'GL', 'SL', 'H2H'] }],
  maxPlayersPerTeam: { type: Number, default: 7 },
  totalPlayers: { type: Number, default: 11 },
  credits: { type: Number, default: 100 }
}, { timestamps: true });

matchSchema.virtual('countdown').get(function() {
  const diff = new Date(this.startTime) - new Date();
  if (diff <= 0) return '0s';
  const d = Math.floor(diff / 86400000);
  const h = Math.floor((diff % 86400000) / 3600000);
  const m = Math.floor((diff % 3600000) / 60000);
  const s = Math.floor((diff % 60000) / 1000);
  let r = '';
  if (d > 0) r += d + 'd ';
  if (h > 0) r += h + 'h ';
  r += m + 'm ' + s + 's';
  return r.trim();
});

matchSchema.set('toJSON', { virtuals: true });
module.exports = mongoose.model('Match', matchSchema);
```

```javascript
const jwt = require('jsonwebtoken');
const User = require('../models/User');

exports.protect = async (req, res, next) => {
  let token;
  if (req.headers.authorization && req.headers.authorization.startsWith('Bearer')) {
    token = req.headers.authorization.split(' ')[1];
  }
  if (!token) {
    return res.status(401).json({ success: false, message: 'Not authorized. Please login.' });
  }
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.user = await User.findById(decoded.id);
    if (!req.user) {
      return res.status(401).json({ success: false, message: 'User not found.' });
    }
    next();
  } catch (error) {
    return res.status(401).json({ success: false, message: 'Token invalid or expired.' });
  }
};

exports.admin = (req, res, next) => {
  if (req.user && req.user.role === 'admin') {
    next();
  } else {
    return res.status(403).json({ success: false, message: 'Admin access required.' });
  }
};
```

## 📄 File: `middleware/errorHandler.js`

```javascript
const errorHandler = (err, req, res, next) => {
  let error = { ...err };
  error.message = err.message;

  if (process.env.NODE_ENV === 'development') {
    console.error('❌ Error:', err.stack);
  }

  if (err.name === 'CastError') {
    error.message = 'Resource not found';
    error.statusCode = 404;
  }
  if (err.code === 11000) {
    error.message = 'Duplicate field value entered';
    error.statusCode = 400;
  }
  if (err.name === 'ValidationError') {
    error.message = Object.values(err.errors).map(v => v.message).join(', ');
    error.statusCode = 400;
  }
  if (err.name === 'JsonWebTokenError') {
    error.message = 'Invalid token';
    error.statusCode = 401;
  }
  if (err.name === 'TokenExpiredError') {
    error.message = 'Token expired';
    error.statusCode = 401;
  }

  res.status(error.statusCode || 500).json({
    success: false,
    message: error.message || 'Server Error'
  });
};

module.exports = errorHandler;
```

```javascript
const jwt = require('jsonwebtoken');
const { validationResult } = require('express-validator');
const User = require('../models/User');

const generateToken = (id) => {
  return jwt.sign({ id }, process.env.JWT_SECRET, { expiresIn: process.env.JWT_EXPIRE || '7d' });
};

exports.register = async (req, res, next) => {
  try {
    const errors = validationResult(req);
    if (!errors.isEmpty()) return res.status(400).json({ success: false, errors: errors.array() });

    const { name, email, password } = req.body;
    const existing = await User.findOne({ email });
    if (existing) return res.status(400).json({ success: false, message: 'Email already exists' });

    const user = await User.create({ name, email, password });
    const token = generateToken(user._id);

    res.status(201).json({
      success: true, message: 'Registration successful', token,
      user: { id: user._id, name: user.name, email: user.email, role: user.role }
    });
  } catch (error) { next(error); }
};

exports.login = async (req, res, next) => {
  try {
    const errors = validationResult(req);
    if (!errors.isEmpty()) return res.status(400).json({ success: false, errors: errors.array() });

    const { email, password } = req.body;
    const user = await User.findOne({ email }).select('+password');
    if (!user) return res.status(401).json({ success: false, message: 'Invalid credentials' });

    const isMatch = await user.comparePassword(password);
    if (!isMatch) return res.status(401).json({ success: false, message: 'Invalid credentials' });

    const token = generateToken(user._id);
    res.json({
      success: true, message: 'Login successful', token,
      user: { id: user._id, name: user.name, email: user.email, role: user.role }
    });
  } catch (error) { next(error); }
};

exports.getMe = async (req, res, next) => {
  try {
    const user = await User.findById(req.user.id).populate('savedMatches');
    res.json({ success: true, user });
  } catch (error) { next(error); }
};

exports.updateProfile = async (req, res, next) => {
  try {
    const { name, email } = req.body;
    const user = await User.findByIdAndUpdate(req.user.id, { name, email }, { new: true, runValidators: true });
    res.json({ success: true, message: 'Profile updated', user });
  } catch (error) { next(error); }
};
```

```javascript
const Team = require('../models/Team');
const Player = require('../models/Player');
const Match = require('../models/Match');
const { generateTeams } = require('../utils/teamGenerator');

exports.generateTeamsForMatch = async (req, res, next) => {
  try {
    const { matchId, format = 'Mega', numberOfTeams = 5 } = req.body;
    if (!matchId) return res.status(400).json({ success: false, message: 'matchId is required' });

    const match = await Match.findById(matchId);
    if (!match) return res.status(404).json({ success: false, message: 'Match not found' });

    const players = await Player.find({ matchId });
    if (players.length < 11) {
      return res.status(400).json({ success: false, message: 'Not enough players' });
    }

    const generatedTeams = generateTeams({
      players, match, format,
      numberOfTeams: Math.min(numberOfTeams, 20)
    });

    const savedTeams = [];
    for (let i = 0; i < generatedTeams.length; i++) {
      const t = generatedTeams[i];
      const team = await Team.create({
        userId: req.user.id, matchId,
        players: t.players, captain: t.captain, viceCaptain: t.viceCaptain,
        totalCredits: t.totalCredits, format,
        teamNumber: i + 1, predictedPoints: t.predictedPoints
      });
      savedTeams.push(team);
    }

    res.status(201).json({
      success: true, message: `${savedTeams.length} teams generated`,
      count: savedTeams.length, format, teams: savedTeams
    });
  } catch (error) { next(error); }
};
```

```javascript
const ROLE_DISTRIBUTIONS = {
  Mega: [
    { WK: 1, BAT: 4, AR: 2, BOWL: 4 },
    { WK: 1, BAT: 3, AR: 3, BOWL: 4 },
    { WK: 2, BAT: 3, AR: 2, BOWL: 4 },
    { WK: 1, BAT: 4, AR: 3, BOWL: 3 },
    { WK: 2, BAT: 4, AR: 2, BOWL: 3 }
  ],
  GL: [
    { WK: 1, BAT: 3, AR: 3, BOWL: 4 },
    { WK: 2, BAT: 3, AR: 3, BOWL: 3 },
    { WK: 1, BAT: 4, AR: 2, BOWL: 4 },
    { WK: 1, BAT: 3, AR: 4, BOWL: 3 },
    { WK: 2, BAT: 4, AR: 1, BOWL: 4 }
  ],
  SL: [
    { WK: 1, BAT: 4, AR: 2, BOWL: 4 },
    { WK: 1, BAT: 3, AR: 3, BOWL: 4 },
    { WK: 2, BAT: 3, AR: 2, BOWL: 4 }
  ],
  H2H: [
    { WK: 1, BAT: 4, AR: 2, BOWL: 4 },
    { WK: 1, BAT: 3, AR: 3, BOWL: 4 },
    { WK: 2, BAT: 4, AR: 2, BOWL: 3 }
  ]
};

function calculatePlayerScore(player) {
  const stats = player.stats || {};
  let score = 0;
  score += (player.form || 50) * 0.4;
  if (['BAT', 'WK', 'AR'].includes(player.role)) {
    score += (stats.average || 0) * 0.15;
    score += (stats.strikeRate || 0) * 0.1;
    score += (stats.runs || 0) * 0.01;
    score += (stats.fifties || 0) * 2;
    score += (stats.hundreds || 0) * 4;
  }
  if (['BOWL', 'AR'].includes(player.role)) {
    score += (stats.wickets || 0) * 3;
    score += (5 - (stats.economy || 8)) * 2;
    score += (stats.fiveWickets || 0) * 5;
  }
  score += (stats.catches || 0) * 0.5;
  score += (stats.stumpings || 0) * 1;
  score += (player.isSelected || 0) * 0.05;
  return Math.round(score * 100) / 100;
}

function shuffleArray(array) {
  const arr = [...array];
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

function selectPlayers(players, distribution, maxCredits, maxPerTeam, team1Short, team2Short) {
  const selected = [];
  let usedCredits = 0;
  const teamCount = { [team1Short]: 0, [team2Short]: 0 };

  const playersByRole = { WK: [], BAT: [], AR: [], BOWL: [] };
  players.forEach(p => {
    if (playersByRole[p.role]) {
      playersByRole[p.role].push({ ...p.toObject(), score: calculatePlayerScore(p) });
    }
  });

  Object.keys(playersByRole).forEach(role => {
    playersByRole[role] = shuffleArray(playersByRole[role])
      .sort((a, b) => (b.score + Math.random() * 10) - (a.score + Math.random() * 10));
  });

  for (const [role, count] of Object.entries(distribution)) {
    const available = playersByRole[role] || [];
    let picked = 0;
    for (const player of available) {
      if (picked >= count) break;
      if (usedCredits + player.credits > maxCredits) continue;
      const pt = player.teamShortName;
      if ((teamCount[pt] || 0) >= maxPerTeam) continue;
      if (selected.find(p => p.playerId.toString() === player._id.toString())) continue;

      selected.push({
        playerId: player._id, name: player.name, role: player.role,
        team: player.team, credits: player.credits, score: player.score,
        isCaptain: false, isViceCaptain: false
      });
      usedCredits += player.credits;
      teamCount[pt] = (teamCount[pt] || 0) + 1;
      picked++;
    }
  }
  return { players: selected, totalCredits: usedCredits };
}

function selectCaptainVC(players) {
  const sorted = [...players].sort((a, b) => b.score - a.score);
  const cIdx = Math.floor(Math.random() * Math.min(3, sorted.length));
  const captain = sorted[cIdx];
  let vcIdx = cIdx === 0 ? 1 : 0;
  if (vcIdx >= sorted.length) vcIdx = sorted.length - 1;
  return { captain, viceCaptain: sorted[vcIdx] };
}

function generateTeams({ players, match, format = 'Mega', numberOfTeams = 5 }) {
  const distributions = ROLE_DISTRIBUTIONS[format] || ROLE_DISTRIBUTIONS.Mega;
  const maxCredits = match.credits || 100;
  const maxPerTeam = match.maxPlayersPerTeam || 7;
  const team1Short = match.team1.shortName;
  const team2Short = match.team2.shortName;
  const generatedTeams = [];
  const usedCombinations = new Set();
  let attempts = 0;

  while (generatedTeams.length < numberOfTeams && attempts < numberOfTeams * 50) {
    attempts++;
    const distribution = distributions[generatedTeams.length % distributions.length];
    const { players: selectedPlayers, totalCredits } = selectPlayers(
      players, distribution, maxCredits, maxPerTeam, team1Short, team2Short
    );

    if (selectedPlayers.length !== 11) continue;

    const comboKey = selectedPlayers.map(p => p.playerId.toString()).sort().join(',');
    if (usedCombinations.has(comboKey)) continue;
    usedCombinations.add(comboKey);

    const { captain, viceCaptain } = selectCaptainVC(selectedPlayers);

    const finalPlayers = selectedPlayers.map(p => ({
      playerId: p.playerId, name: p.name, role: p.role, team: p.team, credits: p.credits,
      isCaptain: p.playerId.toString() === captain.playerId.toString(),
      isViceCaptain: p.playerId.toString() === viceCaptain.playerId.toString()
    }));

    const predictedPoints = Math.round(finalPlayers.reduce((sum, p) => {
      const pd = players.find(pl => pl._id.toString() === p.playerId.toString());
      const base = pd ? calculatePlayerScore(pd) : 30;
      let mult = 1;
      if (p.isCaptain) mult = 2;
      else if (p.isViceCaptain) mult = 1.5;
      return sum + base * mult;
    }, 0));

    generatedTeams.push({
      players: finalPlayers,
      captain: { playerId: captain.playerId, name: captain.name },
      viceCaptain: { playerId: viceCaptain.playerId, name: viceCaptain.name },
      totalCredits, predictedPoints
    });
  }
  return generatedTeams;
}

module.exports = { generateTeams, calculatePlayerScore };
```
